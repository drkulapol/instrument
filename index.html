SQL ตรวจความครบถ้วน ICD9 ↔ Instrument (ADP) ผู้ป่วยใน

สำหรับ HOSxP XE (PostgreSQL) · ตรวจว่า AN ผู้ป่วยในที่มีหัตถการหรืออวัยวะเทียม ลงรหัส ICD9, ADP และ PDx สอดคล้องกันครบตาม 77 กฎ ผลลัพธ์ออกมาเป็น 1 แถวต่อ 1 AN ที่ผิด 1 กฎ พร้อมหมายเหตุและวิธีแก้ไข

วิธีใช้

  1. กดปุ่ม คัดลอกโค้ด
  2. วางใน SQLQueryForm ของ HOSxP
  3. แก้ช่วงวันจำหน่ายที่บรรทัด DATE '2026-10-01' และ DATE '2026-10-31' (รูปแบบ ปี ค.ศ.-เดือน-วัน)
  4. กด Execute แล้ว export ผลเป็น Excel ส่งให้ทีมเวชระเบียน/coder แก้ไข

กลุ่มกฎที่ตรวจ

  • A มี ICD9 หัตถการ แต่ไม่พบ ADP อุปกรณ์ที่ต้องใช้ (เช่น 1371 ต้องมี IOL 2006/2007)
  • B มี ADP อุปกรณ์ แต่ไม่พบ ICD9 หัตถการที่รองรับ (เช่น mesh 5608 ต้องเป็นรหัสไส้เลื่อนแบบ with graft)
  • C ADP ที่ต้องมาเป็นคู่ (เช่น plate 7301 คู่กับ screw 7214B)
  • D ความสอดคล้องกับ PDx และกรณีถอดอุปกรณ์

/* =====================================================================
   ตรวจสอบความครบถ้วน ICD9 ↔ Instrument (ADP) ผู้ป่วยใน (IPD)
   HOSxP XE (PostgreSQL) — rule base = rules_icd9_instrument.csv (76 กฎ)
   ผลลัพธ์ = 1 แถว = 1 AN ที่ผิดเงื่อนไข 1 กฎ พร้อมหมายเหตุ + วิธีแก้ไข
   ---------------------------------------------------------------------
   แหล่งข้อมูล
     ICD9  = iptoprt.icd9            (หัตถการ IPD ที่ส่งออกเคลม)
     ADP   = opitemrece (an) + nondrugitems.nhso_adp_code  (qty สุทธิ > 0)
     PDx   = iptdiag.diagtype = '1'
   แก้ช่วงวันที่ได้ที่ CTE param
   ฉบับเร็ว - เทียบ regex ครั้งเดียวต่อรหัส แล้วสรุปผลต่อ AN
   ใช้เฉพาะไวยากรณ์พื้นฐาน เพื่อให้ SQLQueryForm ของ HOSxP รับได้
   หมายเหตุ - ไฟล์นี้ไม่มีเครื่องหมาย double quote เพราะ HOSxP แปลงเป็น single quote (อิงวันจำหน่าย dchdate)
   ===================================================================== */
WITH
param AS (
  SELECT DATE '2026-10-01' AS d1,
         DATE '2026-10-31' AS d2
),

/* ---------- ประชากร = IPD ที่จำหน่ายในช่วงวันที่ ---------- */
ipd AS (
  SELECT i.an, i.hn, i.regdate, i.dchdate, i.pttype
  FROM ipt i
  CROSS JOIN param p
  WHERE i.dchdate BETWEEN p.d1 AND p.d2
),

/* ---------- ICD9 ที่ลงใน AN ---------- */
icd AS (
  SELECT DISTINCT o.an, replace(upper(trim(o.icd9)), '.', '') AS code
  FROM iptoprt o
  JOIN ipd v ON v.an = o.an
  WHERE coalesce(trim(o.icd9), '') <> ''
),

/* ---------- ADP ที่คีย์ใน AN (กรองช่วงวันที่นอน เพื่อใช้ index rxdate) ---------- */
adp AS (
  SELECT r.an, upper(trim(n.nhso_adp_code)) AS code, sum(r.qty) AS qty
  FROM opitemrece r
  JOIN ipd v          ON v.an = r.an
  JOIN nondrugitems n ON n.icode = r.icode
  WHERE r.rxdate BETWEEN v.regdate AND v.dchdate + 1
    AND coalesce(trim(n.nhso_adp_code), '') <> ''
  GROUP BY r.an, upper(trim(n.nhso_adp_code))
  HAVING sum(r.qty) > 0
),

/* ---------- PDx ---------- */
pdx AS (
  SELECT DISTINCT ON (d.an) d.an, replace(upper(trim(d.icd10)), '.', '') AS code
  FROM iptdiag d
  JOIN ipd v ON v.an = d.an
  WHERE d.diagtype = '1'
  ORDER BY d.an
),

/* ---------- เทียบ regex ครั้งเดียวต่อรหัส สรุปเป็นจำนวนที่ตรงต่อ AN x กฎ ---------- */
f_icd AS (
  SELECT c.an, r.rule_id,
    sum(CASE WHEN r.t_icd9   IS NOT NULL AND c.code ~ r.t_icd9   THEN 1 ELSE 0 END) AS t_i,
    sum(CASE WHEN r.r_icd9_a IS NOT NULL AND c.code ~ r.r_icd9_a THEN 1 ELSE 0 END) AS r_ia,
    sum(CASE WHEN r.r_icd9_b IS NOT NULL AND c.code ~ r.r_icd9_b THEN 1 ELSE 0 END) AS r_ib
  FROM icd c
  CROSS JOIN rules r
  GROUP BY c.an, r.rule_id
),
f_adp AS (
  SELECT c.an, r.rule_id,
    sum(CASE WHEN r.t_adp IS NOT NULL AND c.code ~ r.t_adp THEN 1 ELSE 0 END) AS t_a,
    sum(CASE WHEN r.r_adp IS NOT NULL AND c.code ~ r.r_adp THEN 1 ELSE 0 END) AS r_a
  FROM adp c
  CROSS JOIN rules r
  GROUP BY c.an, r.rule_id
),
f_pdx AS (
  SELECT c.an, r.rule_id,
    sum(CASE WHEN r.r_pdx IS NOT NULL AND c.code ~ r.r_pdx THEN 1 ELSE 0 END) AS r_p
  FROM pdx c
  CROSS JOIN rules r
  GROUP BY c.an, r.rule_id
),

/* ---------- AN ที่มีรหัสอย่างน้อย 1 อย่าง x ทุกกฎ ---------- */
an_rule AS (
  SELECT v.an, r.rule_id
  FROM ipd v
  CROSS JOIN rules r
  WHERE EXISTS (SELECT 1 FROM icd x WHERE x.an = v.an)
     OR EXISTS (SELECT 1 FROM adp x WHERE x.an = v.an)
),

/* ---------- เข้าเงื่อนไขถ้าพบ แต่ไม่ผ่านเงื่อนไขต้องพบ ---------- */
fail AS (
  SELECT ar.an, r.*
  FROM an_rule ar
  JOIN rules r       ON r.rule_id = ar.rule_id
  LEFT JOIN f_icd fi ON fi.an = ar.an AND fi.rule_id = ar.rule_id
  LEFT JOIN f_adp fa ON fa.an = ar.an AND fa.rule_id = ar.rule_id
  LEFT JOIN f_pdx fp ON fp.an = ar.an AND fp.rule_id = ar.rule_id
  WHERE (r.t_icd9 IS NULL OR coalesce(fi.t_i, 0) > 0)
    AND (r.t_adp  IS NULL OR coalesce(fa.t_a, 0) > 0)
    AND NOT (
          (r.r_icd9_a IS NULL OR coalesce(fi.r_ia, 0) > 0)
      AND (r.r_icd9_b IS NULL OR coalesce(fi.r_ib, 0) > 0)
      AND (r.r_adp    IS NULL OR coalesce(fa.r_a,  0) > 0)
      AND (r.r_pdx    IS NULL OR coalesce(fp.r_p,  0) > 0)
    )
),

/* ---------- สรุปรหัสต่อ AN ---------- */
icd_summ AS (
  SELECT an, string_agg(code, ', ' ORDER BY code) AS icd9_list
  FROM icd GROUP BY an
),
adp_summ AS (
  SELECT an, string_agg(code || ' x' || qty, ', ' ORDER BY code) AS adp_list
  FROM adp GROUP BY an
)

/* ---------- ผลลัพธ์ ---------- */
SELECT
  f.an                                              AS an,
  v.hn                                              AS hn,
  concat_ws(' ', concat(p.pname, p.fname), p.lname) AS ชื่อ_สกุล,
  v.dchdate                                         AS วันจำหน่าย,
  concat_ws(' ', v.pttype, pt.name)                 AS สิทธิ,
  d.code                                            AS pdx,
  si.icd9_list                                      AS icd9_ที่ลง,
  sa.adp_list                                       AS adp_ที่ลง,
  f.rule_id                                         AS rule_id,
  f.lvl                                             AS ระดับ,
  f.note                                            AS หมายเหตุ,
  f.fix || CASE left(f.rule_id, 1)
             WHEN 'A' THEN ' | ถ้าไม่ได้ใช้อุปกรณ์จริง ให้ทบทวน ICD9 ให้ตรงกับหัตถการที่ทำ'
             WHEN 'B' THEN ' | ถ้าไม่ได้ใช้อุปกรณ์จริง ให้ลบ/แก้รายการ ADP ในค่าใช้จ่าย'
             WHEN 'C' THEN ' | ถ้าไม่ได้ใช้คู่กันจริง ให้ตรวจว่าคีย์ ADP ถูกรหัส'
             ELSE ''
           END                                      AS วิธีแก้ไข
FROM fail f
JOIN ipd v           ON v.an = f.an
LEFT JOIN pdx d      ON d.an = f.an
LEFT JOIN icd_summ si ON si.an = f.an
LEFT JOIN adp_summ sa ON sa.an = f.an
LEFT JOIN patient p  ON p.hn = v.hn
LEFT JOIN pttype pt  ON pt.pttype = v.pttype
ORDER BY
  CASE f.lvl WHEN 'ERROR' THEN 1 WHEN 'WARNING' THEN 2 ELSE 3 END,
  v.dchdate, f.an, f.rule_id;
